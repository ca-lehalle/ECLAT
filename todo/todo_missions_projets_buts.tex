% --- Une colonne -----------------------------------------------------
%  #1 x de départ   #2 largeur   #3 nombre de lignes   #4 couleur de fond
%  #5 titre EN      #6 titre FR   #7 demi-taille de la case à cocher
%  #8 case de lien (1 = oui)      #9 nombre de lignes d'écriture par ligne
\newcommand{\todocol}[9]{%
  \pgfmathsetmacro{\rhC}{(12.6-1.45)/#3}
  \fill[#4] (#1,-1.45) rectangle ({#1+#2},-12.6);
  \node[anchor=west,align=left,inner sep=0,font=\fontsize{6}{7}\selectfont]
    at ({#1+0.15},-1.1) {\textcolor{gold}{\bfseries #5}\\\textcolor{navy}{\bfseries #6}};
  \foreach \ri in {1,...,#3}{%
    \pgfmathsetmacro{\yt}{-1.45-(\ri-1)*\rhC}
    \pgfmathsetmacro{\yc}{\yt-\rhC/#9+#7+0.07}
    \pgfmathsetmacro{\xe}{#1+#2-0.15-0.7*#8}
    % numéro + case à cocher
    \node[inner sep=0,font=\fontsize{5.5}{6}\selectfont,text=navy!70] at ({#1+0.22},\yc) {\ri};
    \draw[gridgray,line width=0.5pt] ({#1+0.45},{\yc-#7}) rectangle ({#1+0.45+2*#7},{\yc+#7});
    % lignes d'écriture
    \foreach \j in {1,...,#9}{%
      \pgfmathsetmacro{\yy}{\yt-\j*\rhC/#9}
      \ifnum\j=#9
        \draw[gridgray,line width=0.4pt] ({#1+0.45+2*#7+0.12},\yy) -- (\xe,\yy);
      \else
        \draw[gridgray,dotted,line width=0.4pt] ({#1+0.45+2*#7+0.12},\yy) -- (\xe,\yy);
      \fi
    }
    % case de lien vers l'échelle supérieure
    \ifnum#8=1
      \draw[gridgray,line width=0.5pt,rounded corners=1pt]
        ({#1+#2-0.7},{\yc-0.11}) rectangle ({#1+#2-0.15},{\yc+0.11});
    \fi
  }
}
